\documentclass[11pt, a4paper]{awesome-cv}

% Configure page margins with geometry
\geometry{left=1.4cm, top=.8cm, right=1.4cm, bottom=1.8cm, footskip=.5cm}

% Specify the location of the included fonts
\fontdir[fonts/]

% Color for highlights

% Social info separator

%-------------------------------------------------------------------------------

%-------------------------------------------------------------------------------
\name{[FIRST_NAME]}{[LAST_NAME]}

\mobile{[YOUR_PHONE]}

%-------------------------------------------------------------------------------

  \cvskill

    {[SECONDARY_SKILLS_LIST]}

  \cvskill

    {[LANGUAGES_LIST]}

\begin{cventries}

    {[LOCATION_1]} % Location

\cvsection{Education}

\begin{cventries}
  \cventry
    {[DEGREE_1]} % Degree
    {[INSTITUTION_1]} % Institution
    {[LOCATION_EDU_1]} % Location
    {[GRADUATION_DATE_1]} % Date(s)

      \begin{cvitems}
        \item {[THESIS_OR_COURSEWORK_1]}

\begin{cvhonors}
  \cvhonor
    {[CERTIFICATION_NAME]} % Award
    {[ISSUING_ORGANIZATION]} % Event
    {[LOCATION_OR_ONLINE]} % Location
    {[YEAR]} % Date(s)
